dmesg -n 8
dmesg -c >/dev/null

dump(){ printf "%s = " "$1"; devmem "$1" 32; }
stat(){ cat "/sys/class/net/eth0/statistics/$1" 2>/dev/null || echo 0; }
show_stats(){
  echo "tx_packets=$(stat tx_packets)"
  echo "rx_packets=$(stat rx_packets)"
  echo "tx_errors=$(stat tx_errors)"
  echo "rx_errors=$(stat rx_errors)"
  echo "tx_dropped=$(stat tx_dropped)"
  echo "rx_dropped=$(stat rx_dropped)"
  echo "multicast=$(stat multicast)"
  echo "rx_bytes=$(stat rx_bytes)"
  echo "tx_bytes=$(stat tx_bytes)"
}

mib_snapshot(){
  tag="$1"
  echo "=== mib-$tag (0x12c00400..0x12c005fc) ==="
  off=1024
  while [ $off -le 1532 ]; do
    addr=$(printf '0x%08x' $((0x12c00000 + off)))
    printf "%s %s = " "$tag" "$addr"
    devmem "$addr" 32
    off=$((off+4))
  done
}

ip link set br-lan down 2>/dev/null || true
ip link set eth0 nomaster 2>/dev/null || true
ip addr flush dev br-lan 2>/dev/null || true
ip addr flush dev eth0
ip addr add 192.168.77.3/24 dev eth0
ip link set eth0 up
ip link set eth0 promisc on
ip neigh replace 192.168.77.2 lladdr bc:ec:a0:2d:6c:9b nud permanent dev eth0
ip route replace 192.168.77.0/24 dev eth0

echo "=== pre ==="
ip -d link show dev eth0
cat /sys/class/net/eth0/address
cat /sys/class/net/eth0/flags 2>/dev/null || true
show_stats
cat /proc/interrupts | grep -Ei ' 64:| 66:|ERR'
dump 0x12c03804
dump 0x12c03808
dump 0x12c03c44
dump 0x12c02c44
mib_snapshot pre

echo "=== phase1 rx-only observe (24 x 5s) ==="
i=1
while [ $i -le 24 ]; do
  rx0=$(stat rx_packets)
  tx0=$(stat tx_packets)
  rb0=$(stat rx_bytes)
  mb0=$(stat multicast)
  e0=$(stat rx_errors)
  w0=$(dmesg | grep -c 'NETDEV WATCHDOG')
  sleep 5
  rx1=$(stat rx_packets)
  tx1=$(stat tx_packets)
  rb1=$(stat rx_bytes)
  mb1=$(stat multicast)
  e1=$(stat rx_errors)
  w1=$(dmesg | grep -c 'NETDEV WATCHDOG')
  drx=$((rx1-rx0))
  dtx=$((tx1-tx0))
  drb=$((rb1-rb0))
  dmb=$((mb1-mb0))
  derx=$((e1-e0))
  dw=$((w1-w0))
  echo "tick $i/24 rx_delta=$drx tx_delta=$dtx rxbytes_delta=$drb mcast_delta=$dmb rxerr_delta=$derx watchdog_new=$dw rx_now=$rx1 tx_now=$tx1"
  i=$((i+1))
done

echo "post1"
sleep 4
echo "post2"

echo "=== post ==="
ip -d link show dev eth0
cat /sys/class/net/eth0/flags 2>/dev/null || true
show_stats
cat /proc/interrupts | grep -Ei ' 64:| 66:|ERR'
dump 0x12c03804
dump 0x12c03808
dump 0x12c03c44
dump 0x12c02c44
mib_snapshot post

echo "=== watchdog block ==="
dmesg | grep -A8 -E 'NETDEV WATCHDOG|Ring 0 queue 0 status summary' | tail -n 320

echo "=== tx tail ==="
dmesg | grep -E 'tc7200u tx submit' | tail -n 120

echo "done"
