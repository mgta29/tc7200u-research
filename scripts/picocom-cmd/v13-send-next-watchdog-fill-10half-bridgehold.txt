dmesg -n 8
dmesg -c >/dev/null

dump(){ printf "%s = " "$1"; devmem "$1" 32; }
stat(){ cat "/sys/class/net/eth0/statistics/$1" 2>/dev/null || echo 0; }
show_stats(){
  echo "tx_packets=$(stat tx_packets)"
  echo "rx_packets=$(stat rx_packets)"
  echo "tx_errors=$(stat tx_errors)"
  echo "rx_errors=$(stat rx_errors)"
  echo "tx_dropped=$(stat tx_dropped)"
  echo "rx_dropped=$(stat rx_dropped)"
}

ip link set br-lan down 2>/dev/null || true
ip link set eth0 nomaster 2>/dev/null || true
ip addr flush dev br-lan 2>/dev/null || true
ip addr flush dev eth0
ip addr add 192.168.77.3/24 dev eth0
ip link set eth0 up
ip neigh replace 192.168.77.2 lladdr bc:ec:a0:2d:6c:9b nud permanent dev eth0
ip route replace 192.168.77.0/24 dev eth0

echo "=== pre ==="
show_stats
cat /proc/interrupts | grep -Ei ' 64:| 66:|ERR'
dump 0x12c03804
dump 0x12c03808
dump 0x12c03c44
dump 0x12c02c44

echo "=== phase1 controlled tx (6 rounds x 40 one-shot) ==="
i=1
while [ $i -le 6 ]; do
  echo "phase1 round $i/6"
  tx0=$(stat tx_packets)
  rx0=$(stat rx_packets)
  n=1
  while [ $n -le 40 ]; do
    ping -I eth0 -c 1 -W 1 192.168.77.2 >/dev/null 2>&1 || true
    if [ $((n % 10)) -eq 0 ]; then
      echo "burst progress $n/40"
    fi
    n=$((n+1))
  done
  tx1=$(stat tx_packets)
  rx1=$(stat rx_packets)
  dtx=$((tx1-tx0))
  drx=$((rx1-rx0))
  wdc=$(dmesg | grep -c 'NETDEV WATCHDOG')
  echo "phase1 round $i/6 tx_delta=$dtx rx_delta=$drx tx_before=$tx0 tx_after=$tx1 rx_before=$rx0 rx_after=$rx1 watchdog_count=$wdc"
  sleep 1
  i=$((i+1))
done

echo "post1"
sleep 4
echo "post2"

echo "=== post ==="
show_stats
cat /proc/interrupts | grep -Ei ' 64:| 66:|ERR'
dump 0x12c03804
dump 0x12c03808
dump 0x12c03c44
dump 0x12c02c44

echo "=== watchdog block ==="
dmesg | grep -A8 -E 'NETDEV WATCHDOG|Ring 0 queue 0 status summary' | tail -n 320

echo "=== tx tail ==="
dmesg | grep -E 'tc7200u tx submit' | tail -n 160

echo "done"
